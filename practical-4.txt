import java.util.Scanner;

public class ClassAverage {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);

        int total = 0;          
        int grade;             
        int count = 0;         
        double average;       

        System.out.println("Enter student grades between 0 and 100.");
        System.out.println("Enter -1 to finish input and calculate average.\n");
        System.out.print("Enter grade (or -1 to quit): ");
        grade = input.nextInt();

        while (grade != -1) {
            if (grade >= 0 && grade <= 100) {
                total += grade;   
                count++;         
            } else {
                System.out.println("Invalid grade! Please enter a number between 0 and 100.");
            }

            System.out.print("Enter next grade (or -1 to quit): ");
            grade = input.nextInt();
        }

        if (count != 0) {
            average = (double) total / count;
            System.out.println("\nNumber of students: " + count);
            System.out.println("Total of all grades: " + total);
            System.out.println("Class average: " + average);
        } else {
            System.out.println("\nNo grades were entered.");
        }

        input.close();
    }
}
